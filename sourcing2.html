/* ═══════════════ PDF EXPORT (uses jsPDF directly for true download) ═════════ */ function exportPDF(){ const btn = document.getElementById('exportBtn'); btn.textContent = '⏳ Generating…'; btn.disabled = true; setTimeout(async () => { try { const element = document.querySelector('.shell'); const month = document.getElementById('fMonth').value; const site = document.getElementById('fSite').value; const fname = `Survey_Dashboard_${month}_${site}_${new Date().getTime()}.pdf`; // Use html2canvas to render the element, then jsPDF to save const canvas = await html2canvas(element, { scale: 2, logging: false, useCORS: true, backgroundColor: '#ffffff' }); const imgData = canvas.toDataURL('image/jpeg', 0.98); const imgWidth = 210; // A4 width in mm const imgHeight = (canvas.height * imgWidth) / canvas.width; const { jsPDF } = window.jspdf; const pdf = new jsPDF('p', 'mm', 'a4'); let heightLeft = imgHeight; let position = 0; // Add pages as needed while (heightLeft >= 0) { pdf.addImage(imgData, 'JPEG', 0, position, imgWidth, imgHeight); heightLeft -= 297; // A4 height in mm position -= 297; if (heightLeft > 0) { pdf.addPage(); } } // Force download without dialog pdf.save(fname); btn.textContent = '✓ PDF Exported'; setTimeout(() => { btn.textContent = '📥 Export PDF'; btn.disabled = false; }, 2000); } catch (err) { console.error('PDF export failed:', err); btn.textContent = '❌ Export Failed'; setTimeout(() => { btn.textContent = '📥 Export PDF'; btn.disabled = false; }, 2000); } }, 100); }
